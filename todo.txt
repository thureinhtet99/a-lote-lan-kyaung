"use server";

import { db } from "@/drizzle/db";
import { usersTable } from "@/drizzle/schema";
import { eq } from "drizzle-orm";
import { insertUser } from "@/features/users/db/users";

type UserData = {
  id: string;
  firstName: string;
  lastName: string;
  username: string;
  email: string;
  imageUrl: string;
};

export async function insertUserToDatabase(userData: UserData) {
  try {
    // First check if the user already exists
    const existingUser = await db.select().from(usersTable).where(eq(usersTable.id, userData.id));
    
    // If the user doesn't exist, insert them
    if (existingUser.length === 0) {
      await insertUser({
        id: userData.id,
        first_name: userData.firstName,
        last_name: userData.lastName,
        username: userData.username,
        email: userData.email,
        image: userData.imageUrl,
        createdAt: new Date(),
        updatedAt: new Date(),
      });
      
      return { success: true, message: "User inserted successfully" };
    }
    
    // User already exists
    return { success: true, message: "User already exists" };
  } catch (error) {
    console.error("Error inserting user:", error);
    return { 
      success: false, 
      message: error instanceof Error ? error.message : "Unknown error occurred" 
    };
  }
}
